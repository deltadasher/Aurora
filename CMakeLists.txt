cmake_minimum_required(VERSION 3.21)
project(AuroraNotes VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Svg)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(aurora-notes src/main.cpp src/NotesStore.cpp src/NotesStore.h src/DocumentHandler.cpp src/DocumentHandler.h)
file(GLOB AURORA_QML CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} qml/*.qml qml/qmldir)
qt_add_resources(aurora-notes "aurora_ui" PREFIX "/" FILES ${AURORA_QML})
file(GLOB AURORA_FONTS CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} fonts/*.ttf)
qt_add_resources(aurora-notes "aurora_fonts" PREFIX "/" FILES ${AURORA_FONTS})
target_link_libraries(aurora-notes PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
include(GNUInstallDirs)
install(TARGETS aurora-notes RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 QUIET COMPONENTS Test)
endif()
if(BUILD_TESTING AND TARGET Qt6::Test)
    qt_add_executable(tst_notesstore tests/tst_notesstore.cpp src/NotesStore.cpp src/NotesStore.h)
    target_include_directories(tst_notesstore PRIVATE src)
    target_link_libraries(tst_notesstore PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
    add_test(NAME notesstore COMMAND tst_notesstore)
    set_tests_properties(notesstore PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

if(BUILD_TESTING AND TARGET Qt6::Test)
    qt_add_executable(aurora-interface-tests tests/tst_interface.cpp src/NotesStore.cpp src/NotesStore.h src/DocumentHandler.cpp src/DocumentHandler.h)
    qt_add_resources(aurora-interface-tests "aurora_test_ui" PREFIX "/" FILES ${AURORA_QML})
    qt_add_resources(aurora-interface-tests "aurora_test_fonts" PREFIX "/" FILES ${AURORA_FONTS})
    target_include_directories(aurora-interface-tests PRIVATE src)
    target_link_libraries(aurora-interface-tests PRIVATE Qt6::Test Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
    add_test(NAME interface COMMAND aurora-interface-tests)
    set_tests_properties(interface PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-config")
endif()
